 42    """
 43    K-SVD dictionary learner.
 44
 45    Alternates between:
 46      1. Sparse coding — encode each training signal over the current D.
 47      2. Dictionary update — update each atom (and its coefficients) via a
 48         rank-1 approximation of the residual matrix.
 49
 50    Parameters
 51    ----------
 52    n_components : int
 53        Number of dictionary atoms this instance learns. When ``D_frozen``
 54        is supplied to ``fit()``, this is the count of *new* atoms only —
 55        the frozen atoms are additional, held-constant columns and are not
 56        counted here.
 57    n_nonzero_coefs : int
 58        Sparsity target T: each signal is represented with at most T atoms.
 59    n_iter : int
 60        Number of K-SVD iterations (default 10).
 61    exact_svd : bool
 62        If True, use full SVD for the atom update (exact K-SVD).
 63        If False (default), use the faster approximate update.
 64    mu_thresh : float
 65        Mutual-incoherence threshold in (0, 1].  Atoms whose pairwise
 66        correlation exceeds this value are replaced.  Set to 1.0 to
 67        disable (default 0.99).
 68    mem_usage : str
 69        One of 'high', 'normal' (default), 'low'.
 70        Controls whether G = D'D (and DtX = D'X) are precomputed.
 71    random_state : int or None
 72        Seed for reproducible atom initialisation.
 73    verbose : bool
 74        Print iteration progress (default False).
 75
 76    Attributes
 77    ----------
 78    D_ : np.ndarray, shape (n_features, n_frozen + n_components)
 79        Learned dictionary (set after fit()). Includes any ``D_frozen``
 80        columns passed to ``fit()``, unchanged, as its leading columns.
 81    Gamma_ : np.ndarray, shape (n_frozen + n_components, n_samples)
 82        Sparse codes for the training data from the final iteration
 83        (set after fit()). Exposed so callers that need the training
 84        codes (e.g. LC-KSVD, which reuses this class on an augmented

365        n_frozen: int,

377            n_samples = int(data["n_samples"])
378            n_components = int(data["n_components"])

380            n_iter = int(data["n_iter"])
381            completed_iter = int(data["completed_iter"])
382            n_frozen_ckpt = int(data["n_frozen"]) if "n_frozen" in data else 0

384            if (n_features, n_samples) != X.shape:

387                    f"{(n_features, n_samples)}, but X has shape {X.shape}."
388                )
389            if n_components != self.n_components:
390                raise DictionaryLearningError(

392                    f"KSVD.n_components={self.n_components}."
393                )
394            if n_nonzero_coefs != self.n_nonzero_coefs:
395                raise DictionaryLearningError(

398                )

414
415            D = data["D"].copy()

417            unused = data["unused"].copy()
418            replaced = data["replaced"].copy()
419            errors_ = list(data["errors_"])
420

265    def fit(
266        self,
267        X: np.ndarray,
268        H: np.ndarray,
269        D_init: np.ndarray | None = None,
270        A_init: np.ndarray | None = None,
271        W_init: np.ndarray | None = None,
272        Q: np.ndarray | None = None,
273        checkpoint_dir: str | None = None,
274        resume: bool = True,
275    ) -> "LCKSVD":
276        """
277        Learn a discriminative dictionary from labelled training data.
278
279        Parameters
280        ----------
281        X : np.ndarray, shape (n_features, n_samples)
282            Training signals.
283        H : np.ndarray, shape (n_classes, n_samples)
284            One-hot label matrix.
285        D_init : np.ndarray or None
286            Initial dictionary. If None, a K-SVD initialisation is run.
287            Ignored when resuming from an existing checkpoint.
288        A_init : np.ndarray or None
289            Initial label-consistency transform. Ignored when resuming.
290        W_init : np.ndarray or None
291            Initial classifier weights (used to warm-start LC-KSVD2's
292            joint optimisation only; LC-KSVD1 trains its classifier
293            separately at the end regardless of this argument). Ignored
294            when resuming.
295        Q : np.ndarray or None
296            Label-consistent target matrix. Computed from H if None.
297            Ignored when resuming.
298        checkpoint_dir : str or None
299            If given, a checkpoint of the *outer* LC-KSVD loop is written
300            to ``<checkpoint_dir>/lc_ksvd_checkpoint.npz`` after every
301            outer iteration, overwriting the previous one. The directory
302            is created if it does not exist. This is independent of, and
303            not passed down to, the inner per-iteration KSVD instance.
304        resume : bool
305            If True (default) and a checkpoint is found in
306            ``checkpoint_dir``, training resumes from it. If False, any
307            existing checkpoint in ``checkpoint_dir`` is ignored and
308            overwritten.
309
310        Returns
311        -------
312        self
313        """
314        X = np.asarray(X, dtype=float)
315        H = np.asarray(H, dtype=float)
316        n_features, n_samples = X.shape
317        n_classes = H.shape[0]
318
319        if n_samples < self.n_components:
320            raise DictionaryLearningError(
321                f"n_samples={X.shape[1]} is less than n_components={self.n_components}. "
322            )
323
324        checkpoint_path = None
325        start_iter = 0
326        D = A = W = Q_loaded = None
327        self.errors_ = []
328
329        if checkpoint_dir is not None:
330            os.makedirs(checkpoint_dir, exist_ok=True)
331            checkpoint_path = os.path.join(checkpoint_dir, _CHECKPOINT_FILENAME)
332
333            if resume and os.path.exists(checkpoint_path):
334                (
335                    D, A, W, Q_loaded, self.errors_, start_iter,
336                ) = self._load_checkpoint(checkpoint_path, X, H)
337                if self.verbose:
338                    print(
339                        f"Resuming from checkpoint at outer iteration "
340                        f"{start_iter}/{self.n_iter} ({checkpoint_path})"
341                    )
342
343        # ---- Initialisation (skipped if resuming from a checkpoint) ----
344        if D is None:
345            if D_init is None or A_init is None or W_init is None or Q is None:
346                if self.verbose:
347                    print("Running initialisation K-SVD...")
348                D_init, A_init, W_init, Q = initialization4lcksvd(
349                    X, H,
350                    self.n_components,
351                    self.n_iter_init,
352                    self.n_nonzero_coefs,
353                    random_state=self.random_state,
354                    verbose=self.verbose,
355                    lambda1=self.lambda1,
356                    lambda2=self.lambda2,
357                )
358
359            D = normalize_columns(D_init.copy())
360            A = A_init.copy()
361            W = W_init.copy()
362        else:
363            Q = Q_loaded
364
365        sqrt_alpha = self.alpha
366        sqrt_beta = self.beta
367
368        use_classifier_term = (self.variant == "lcksvd2")
369
370        # ---- Build augmented training data ----
371        # Y_aug = [X ; sqrt_alpha*Q ; sqrt_beta*H]  (LC-KSVD2)
372        # Y_aug = [X ; sqrt_alpha*Q]                 (LC-KSVD1)
373        H_aug = H if use_classifier_term else None
374        X_aug, _, _ = _augment_data(X, Q, H_aug, sqrt_alpha, sqrt_beta)
375
376        # A single K-SVD instance, reused every outer iteration. Only
377        # D_init changes between calls (n_iter=1 each time), so the atom
378        # update / incoherence-clearing logic lives in exactly one place.
379        atom_updater = KSVD(
380            n_components=self.n_components,
381            n_nonzero_coefs=self.n_nonzero_coefs,
382            n_iter=1,
383            exact_svd=self.exact_svd,
384            mu_thresh=self.mu_thresh,
385            mem_usage="normal",
386            random_state=self.random_state,
387            verbose=False,
388        )
389
390        Gamma = None
391        for it in range(start_iter, self.n_iter):
392
393            # ---- Build augmented dictionary ----
394            # D_aug = [D ; sqrt_alpha*A ; sqrt_beta*W]
395            D_aug = self._build_aug_dict(D, A, W, sqrt_alpha, sqrt_beta, use_classifier_term)
396            D_aug_norm = normalize_columns(D_aug)
397
398            # ---- Sparse code + single atom-update pass on the augmented
399            #      system, delegated to KSVD ----
400            atom_updater.fit(X_aug, D_init=D_aug_norm, checkpoint_dir=None)
401            D_aug_updated = atom_updater.D_
402            Gamma = atom_updater.Gamma_
403
404            # De-augment: extract D, A, W from the updated augmented dict
405            D, A, W = self._split_aug_dict(
406                D_aug_updated, n_features, n_classes, sqrt_alpha, sqrt_beta, use_classifier_term
407            )
408            D = normalize_columns(D)
409
410            # ---- Track RMSE on original X ----
411            err = float(np.sqrt(rep_error_squared(X, D, Gamma).sum() / X.size))
412            self.errors_.append(err)
413
414            if self.verbose:
415                print(f"[{self.variant.upper()}] Iter {it + 1}/{self.n_iter}  RMSE={err:.6f}")
416
417            if checkpoint_path is not None:
418                self._save_checkpoint(
419                    checkpoint_path, X, H, D, A, W, Q, it + 1
420                )
421
422        self.D_ = D
423        self.A_ = A
424
425        # ---- Classifier ----
426        if self.variant == "lcksvd1":
427            # Paper Sec. 3.2: LC-KSVD1's classifier is trained separately,
428            # after D, A, X are computed — never jointly with the
429            # dictionary. Re-encode with the final, converged D.
430            clf = self.classifier if self.classifier is not None else RidgeClassifier(
431                lambda1=self.lambda1
432            )
433            Gamma_final = self.transform(X) if self.D_ is not None else Gamma
434            clf.fit(Gamma_final, H)
435            self.classifier_ = clf
436            # Mirror a linear W_ for convenience/back-compat when the
437            # fitted classifier exposes one (e.g. the default
438            # RidgeClassifier); None for classifiers that don't.
439            self.W_ = getattr(clf, "W_", None)
440        else:
441            self.W_ = W
442            self.classifier_ = None
443
444        # Record per-class atom ranges matching _build_label_consistent_target
445        atoms_per_class = self.n_components // n_classes
446        boundaries: dict[int, tuple[int, int]] = {}
447        for c in range(n_classes):
448            start = c * atoms_per_class
449            end = start + atoms_per_class if c < n_classes - 1 else self.n_components
450            boundaries[c] = (start, end)
451        self.class_boundaries_ = boundaries
452
453        return self

459    def _save_checkpoint(
460        self,
461        path: str,
462        X: np.ndarray,
463        H: np.ndarray,
464        D: np.ndarray,
465        A: np.ndarray,
466        W: np.ndarray,
467        Q: np.ndarray,
468        completed_iter: int,
469    ) -> None:
470        """
471        Atomically write the outer LC-KSVD training state to ``path``.
472
473        Written to a temp file in the same directory first, then moved
474        into place with os.replace, so an abrupt stop mid-write can never
475        leave a corrupt/truncated checkpoint at ``path``.
476        """
477        directory = os.path.dirname(path) or "."
478        fd, tmp_path = tempfile.mkstemp(
479            dir=directory, prefix=".lc_ksvd_checkpoint_", suffix=".npz.tmp"
480        )
481        try:
482            # See KSVD._save_checkpoint: np.savez silently appends '.npz'
483            # to string paths not already ending in exactly that
484            # extension, which would orphan the real data under a
485            # mangled filename and leave an empty file at `path` after
486            # os.replace. Writing through the fd avoids that entirely.
487            with os.fdopen(fd, "wb") as f:
488                np.savez(
489                    f,
490                    D=D,
491                    A=A,
492                    W=W,
493                    Q=Q,
494                    errors_=np.asarray(self.errors_, dtype=float),
495                    completed_iter=completed_iter,
496                    n_iter=self.n_iter,
497                    n_components=self.n_components,
498                    n_nonzero_coefs=self.n_nonzero_coefs,
499                    alpha=self.alpha,
500                    beta=self.beta,
501                    variant=self.variant,
502                    n_features=X.shape[0],
503                    n_samples=X.shape[1],
504                    n_classes=H.shape[0],
505                )
506            os.replace(tmp_path, path)
507        finally:
508            if os.path.exists(tmp_path):
509                os.remove(tmp_path)

511    def _load_checkpoint(self, path: str, X: np.ndarray, H: np.ndarray):
512        """
513        Load and validate a checkpoint against the current config, X, and H.
514
515        Raises DictionaryLearningError on any mismatch, rather than
516        silently resuming with an incompatible state.
517        """
518        with np.load(path, allow_pickle=True) as data:
519            n_features = int(data["n_features"])
520            n_samples = int(data["n_samples"])
521            n_classes = int(data["n_classes"])
522            n_components = int(data["n_components"])
523            n_nonzero_coefs = int(data["n_nonzero_coefs"])
524            n_iter = int(data["n_iter"])
525            alpha = float(data["alpha"])
526            beta = float(data["beta"])
527            variant = str(data["variant"])
528            completed_iter = int(data["completed_iter"])
529
530            if (n_features, n_samples) != X.shape:
531                raise DictionaryLearningError(
532                    f"Checkpoint at {path} was computed on data of shape "
533                    f"{(n_features, n_samples)}, but X has shape {X.shape}."
534                )
535            if n_classes != H.shape[0]:
536                raise DictionaryLearningError(
537                    f"Checkpoint n_classes={n_classes} does not match "
538                    f"H.shape[0]={H.shape[0]}."
539                )
540            if n_components != self.n_components:
541                raise DictionaryLearningError(
542                    f"Checkpoint n_components={n_components} does not match "
543                    f"LCKSVD.n_components={self.n_components}."
544                )
545            if n_nonzero_coefs != self.n_nonzero_coefs:
546                raise DictionaryLearningError(
547                    f"Checkpoint n_nonzero_coefs={n_nonzero_coefs} does not "
548                    f"match LCKSVD.n_nonzero_coefs={self.n_nonzero_coefs}."
549                )
550            if variant != self.variant:
551                raise DictionaryLearningError(
552                    f"Checkpoint was created with variant='{variant}', but "
553                    f"this LCKSVD instance has variant='{self.variant}'."
554                )
555            if not np.isclose(alpha, self.alpha) or not np.isclose(beta, self.beta):
556                raise DictionaryLearningError(
557                    f"Checkpoint was created with alpha={alpha}, beta={beta}, "
558                    f"but this LCKSVD instance has alpha={self.alpha}, "
559                    f"beta={self.beta}."
560                )
561            if n_iter != self.n_iter:
562                raise DictionaryLearningError(
563                    f"Checkpoint was created with n_iter={n_iter}, but this "
564                    f"LCKSVD instance has n_iter={self.n_iter}."
565                )
566            if completed_iter >= self.n_iter:
567                raise DictionaryLearningError(
568                    f"Checkpoint at {path} already completed all "
569                    f"{self.n_iter} outer iterations; nothing to resume."
570                )
571
572            D = data["D"].copy()
573            A = data["A"].copy()
574            W = data["W"].copy()
575            Q = data["Q"].copy()
576            errors_ = list(data["errors_"])
577
578        return D, A, W, Q, errors_, completed_iter

596    def predict(self, X: np.ndarray) -> np.ndarray:
597        """
598        Classify test signals using the learned classifier.
599
600        For LC-KSVD2, this is the argmax of ``W_ @ gamma``. For LC-KSVD1,
601        this delegates to the fitted ``classifier_`` (default
602        ``RidgeClassifier``, or a user-supplied ``classifier``).
603
604        Parameters
605        ----------
606        X : np.ndarray, shape (n_features, n_samples)
607
608        Returns
609        -------
610        labels : np.ndarray, shape (n_samples,)  integer class indices
611        """
612        self._check_fitted()
613        Gamma = self.transform(X)
614
615        if self.variant == "lcksvd1":
616            if self.classifier_ is None:
617                raise DictionaryLearningError(
618                    "No classifier is available. This should not happen "
619                    "for a successfully fit LC-KSVD1 model."
620                )
621            return self.classifier_.predict(Gamma)
622
623        if self.W_ is None:
624            raise DictionaryLearningError(
625                "Classifier W is not available. "
626                "Access sparse codes via transform() instead."
627            )
628        scores = self.W_ @ Gamma          # (n_classes, n_samples)
629        return np.argmax(scores, axis=0)

631    def score(self, X: np.ndarray, H: np.ndarray) -> float:
632        """
633        Classification accuracy on (X, H).
634
635        Parameters
636        ----------
637        X : np.ndarray, shape (n_features, n_samples)
638        H : np.ndarray, shape (n_classes, n_samples) — one-hot labels
639
640        Returns
641        -------
642        accuracy : float in [0, 1]
643        """
644        true_labels = np.argmax(H, axis=0)
645        pred_labels = self.predict(X)
646        return float(np.mean(pred_labels == true_labels))

648    @staticmethod
649    def _build_aug_dict(
650        D: np.ndarray,
651        A: np.ndarray,
652        W: np.ndarray,
653        sqrt_alpha: float,
654        sqrt_beta: float,
655        use_classifier: bool,
656    ) -> np.ndarray:
657        """Stack [D ; sqrt_alpha*A ; (sqrt_beta*W)]."""
658        parts = [D, sqrt_alpha * A]
659        if use_classifier:
660            parts.append(sqrt_beta * W)
661        return np.vstack(parts)
662
663    @staticmethod
664    def _split_aug_dict(
665        D_aug: np.ndarray,
666        n_features: int,
667        n_classes: int,
668        sqrt_alpha: float,
669        sqrt_beta: float,
670        use_classifier: bool,
671    ) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
672        """
673        Recover (D, A, W) from the augmented dictionary D_aug.
674
675        D_aug rows are: n_features | n_components | (n_classes if lcksvd2).
676
677        D_aug's columns are unit-norm as a whole (across all stacked
678        blocks), not block-by-block. To recover a properly normalised D
679        together with A/W on a consistent per-atom scale, each atom's D
680        sub-block norm is computed and used to rescale D, A, and W
681        (in addition to removing the sqrt_alpha / sqrt_beta weighting
682        from A / W).
683        """
684        n_components = D_aug.shape[1]
685        D = D_aug[:n_features, :]
686        A_rows = n_components
687        A = D_aug[n_features: n_features + A_rows, :]
688        if use_classifier:
689            W = D_aug[n_features + A_rows:, :]
690        else:
691            W = np.zeros((n_classes, n_components))
692
693        l2norms = np.linalg.norm(D, axis=0)
694        l2norms = np.where(l2norms > 1e-14, l2norms, 1e-14)
695
696        D = D / l2norms
697        A = A / l2norms / max(sqrt_alpha, 1e-14)
698        if use_classifier:
699            W = W / l2norms / max(sqrt_beta, 1e-14)
700
701        return D, A, W

464        """
465        Freeze existing W columns; learn only the columns for new atoms.
466
467        This implements the ``freeze_classifier=True`` behaviour: old class
468        boundaries in W stay fixed; only weights for the n_new_atoms are
469        updated for the new class.

475        n_atoms_old = self.W_.shape[1]

477        # Extend W with zero rows for any new classes and zero cols for new atoms
478        W_extended = np.zeros((n_classes_new, n_atoms_old + n_new_atoms))
479        W_extended[:n_classes_old, :n_atoms_old] = self.W_
480
481        # Learn only the new columns via least squares restricted to new atoms
482        # Encode all accumulated data over full dict, extract new-atom codes
483        X_all = np.hstack(self._X_all)
484        Gamma_full = self._encode(X_all)
485        Gamma_new = Gamma_full[n_atoms_old:, :]   # (n_new_atoms, n_samples)
486
487        # Solve W_new @ Gamma_new ≈ H - W_old @ Gamma_old
488        Gamma_old = Gamma_full[:n_atoms_old, :]
489        residual_H = H - W_extended[:, :n_atoms_old] @ Gamma_old
490        W_new_cols = residual_H @ np.linalg.pinv(Gamma_new)
491        W_extended[:, n_atoms_old:] = W_new_cols
492
493        self.W_ = W_extended

Checkpointing

Each call to fit_base or add_class represents one stage of the incremental pipeline, and each stage trains its own, differently shaped, inner dictionary-learner instance. If a checkpoint_dir is given, this class therefore does NOT hand every stage the same path — doing so would cause the second stage's checkpoint load to fail (or, worse, be silently wrong) since its X shape differs from the first stage's. Instead, each stage gets its own subdirectory:

<checkpoint_dir>/base/           (fit_base)
<checkpoint_dir>/class_<label>/  (add_class, per class_label)

so that interrupting and resuming an individual stage's training does not collide with any other stage's saved state.
